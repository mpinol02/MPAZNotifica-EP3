/**
 * Muestra el mensaje de apoyo de un campo.
 * Si no existe un mensaje, no ocupa espacio adicional.
 */
fun apoyo(mensaje: String?): (@Composable () -> Unit)? {

    if (mensaje == null) return null

    return {
        Text(mensaje)
    }
}

/**
 * Etiqueta reutilizable para tipos, alcances,
 * estados y canales de comunicación.
 */
@Composable
fun EtiquetaInfo(
    texto: String,
    icono: ImageVector,
    modifier: Modifier = Modifier,
    colorFondo: Color = MaterialTheme.colorScheme.secondaryContainer,
    colorTexto: Color = MaterialTheme.colorScheme.onSecondaryContainer
) {

    Surface(
        modifier = modifier,
        shape = RoundedCornerShape(50),
        color = colorFondo,
        contentColor = colorTexto
    ) {

        Row(
            modifier = Modifier.padding(
                horizontal = 10.dp,
                vertical = 4.dp
            ),
            verticalAlignment = Alignment.CenterVertically
        ) {

            Icon(
                imageVector = icono,
                contentDescription = null,
                modifier = Modifier.size(14.dp)
            )

            Spacer(
                modifier = Modifier.width(6.dp)
            )

            Text(
                text = texto,
                style = MaterialTheme.typography.labelMedium
            )
        }
    }
}

/**
 * Tarjeta reutilizable para avisos informativos
 * y mensajes de error.
 */
@Composable
fun AvisoCard(
    texto: String,
    icono: ImageVector,
    modifier: Modifier = Modifier,
    esError: Boolean = false
) {

    // Selecciona los colores según el tipo de aviso.
    val fondo = if (esError) {
        MaterialTheme.colorScheme.errorContainer
    } else {
        MaterialTheme.colorScheme.tertiaryContainer
    }

    val contenido = if (esError) {
        MaterialTheme.colorScheme.onErrorContainer
    } else {
        MaterialTheme.colorScheme.onTertiaryContainer
    }

    Card(
        modifier = modifier.fillMaxWidth(),
        colors = CardDefaults.cardColors(
            containerColor = fondo,
            contentColor = contenido
        )
    ) {

        Row(
            modifier = Modifier.padding(12.dp),
            verticalAlignment = Alignment.CenterVertically,
            horizontalArrangement = Arrangement.spacedBy(12.dp)
        ) {

            Icon(
                imageVector = icono,
                contentDescription = null
            )

            Text(
                text = texto,
                style = MaterialTheme.typography.bodyMedium
            )
        }
    }
}

/**
 * Componente que se muestra cuando una lista
 * no contiene elementos.
 */
@Composable
fun EstadoVacio(
    icono: ImageVector,
    titulo: String,
    detalle: String,
    modifier: Modifier = Modifier
) {

    Column(
        modifier = modifier
            .fillMaxWidth()
            .padding(32.dp),

        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.spacedBy(8.dp)
    ) {

        Icon(
            imageVector = icono,
            contentDescription = null,
            modifier = Modifier.size(48.dp),
            tint = MaterialTheme.colorScheme.onSurfaceVariant
        )

        Text(
            text = titulo,
            style = MaterialTheme.typography.titleMedium,
            textAlign = TextAlign.Center
        )

        Text(
            text = detalle,
            style = MaterialTheme.typography.bodyMedium,
            color = MaterialTheme.colorScheme.onSurfaceVariant,
            textAlign = TextAlign.Center
        )
    }
}

/**
 * Devuelve el icono correspondiente a cada canal.
 */
fun CanalDespacho.icono(): ImageVector = when (this) {

    CanalDespacho.EMAIL -> Icons.Filled.Email

    CanalDespacho.SMS -> Icons.Filled.Sms

    // EV3: utiliza el icono actualizado de Compose.
    CanalDespacho.WHATSAPP -> Icons.AutoMirrored.Filled.Chat
}

/**
 * Informa el estado de la conexión con el servidor.
 *
 * Permite mostrar carga, actualización,
 * errores y acciones de reintento.
 */
@Composable
fun AvisoConexion(
    estado: EstadoDatos,
    sinSesionRemota: Boolean,
    onReintentar: () -> Unit,
    modifier: Modifier = Modifier
) {

    when (estado) {

        EstadoDatos.Inactivo -> Unit

        EstadoDatos.Cargando -> {

            Column(
                modifier = modifier.fillMaxWidth(),
                verticalArrangement = Arrangement.spacedBy(4.dp)
            ) {

                LinearProgressIndicator(
                    modifier = Modifier.fillMaxWidth()
                )

                Text(
                    text = "Actualizando datos desde el servidor…",
                    style = MaterialTheme.typography.labelMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant
                )
            }
        }

        is EstadoDatos.Actualizado -> {

            Row(
                modifier = modifier.fillMaxWidth(),
                verticalAlignment = Alignment.CenterVertically
            ) {

                Icon(
                    imageVector = Icons.Filled.CloudDone,
                    contentDescription = null,
                    modifier = Modifier.size(18.dp),
                    tint = MaterialTheme.colorScheme.primary
                )

                Text(
                    text = "  Datos actualizados ${estado.hora}",
                    style = MaterialTheme.typography.labelMedium,
                    color = MaterialTheme.colorScheme.onSurfaceVariant,
                    modifier = Modifier.weight(1f)
                )

                IconButton(
                    onClick = onReintentar
                ) {

                    Icon(
                        Icons.Filled.Refresh,
                        contentDescription = "Actualizar datos"
                    )
                }
            }
        }

        is EstadoDatos.SinConexion -> {

            Column(
                modifier = modifier.fillMaxWidth()
            ) {

                AvisoCard(
                    texto = estado.mensaje,
                    icono = Icons.Filled.CloudOff
                )

                if (!sinSesionRemota) {

                    TextButton(
                        onClick = onReintentar
                    ) {
                        Text("Reintentar")
                    }
                }
            }
        }

        is EstadoDatos.Error -> {

            Column(
                modifier = modifier.fillMaxWidth()
            ) {

                AvisoCard(
                    texto = estado.mensaje,
                    icono = Icons.Filled.CloudOff,
                    esError = true
                )

                TextButton(
                    onClick = onReintentar
                ) {
                    Text("Reintentar")
                }
            }
        }
    }
}
